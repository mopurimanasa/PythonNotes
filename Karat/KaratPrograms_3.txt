Decode String: Decode patterns like "3[a2[c]]" -> "accaccacc".

def decode_string(s):
    num_stack = []
    str_stack = []

    num = 0
    text = ""

    for ch in s:

        # If character is a number
        if ch.isdigit():
            num = num * 10 + int(ch)

        # If opening bracket comes, save current values
        elif ch == '[':
            num_stack.append(num)
            str_stack.append(text)

            num = 0
            text = ""

        # If closing bracket comes, decode
        elif ch == ']':
            repeat = num_stack.pop()
            previous = str_stack.pop()

            text = previous + text * repeat

        # If character is a letter
        else:
            text = text + ch

    return text


# Input
s = "3[a2[c]]"

# Call the function
result = decode_string(s)

# Output
print(result)

6) LRU Cache: Implement get/put with LRU eviction.

LRU = Least Recently Used.

Suppose the cache can hold only 2 items:

put(1, "A")
put(2, "B")

Cache: {1: A, 2: B}

Now:

get(1)

We just used key 1, so 1 becomes the most recently used.

Least recent → 2
Most recent  → 1

If we now do:

put(3, "C")

the cache is full, so we remove 2, because it was used least recently.

Result:

{1: A, 3: C}

Program:

from collections import OrderedDict


class LRUCache:

    def __init__(self, capacity):
        self.capacity = capacity
        self.cache = OrderedDict()

    def put(self, key, value):

        # If key already exists, move it to the end
        if key in self.cache:
            self.cache.move_to_end(key)

        # Add or update key and value
        self.cache[key] = value

        # If cache exceeds capacity, remove LRU item
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)

    def get(self, key):

        # If key doesn't exist
        if key not in self.cache:
            return -1

        # Key is recently used, so move it to the end
        self.cache.move_to_end(key)

        return self.cache[key]


# Create cache with capacity 2
cache = LRUCache(2)

# PUT operations
cache.put(1, "A")
cache.put(2, "B")

# GET operation
print(cache.get(1))

# Add another item
cache.put(3, "C")

# GET operations
print(cache.get(2))
print(cache.get(3))

7) Flood Fill: Recolor connected pixels starting at (sr, sc):

1) Get the starting color
2) compare old and new color and return image
3) define a function
4) Stop if we go outside the grid
5) Stop if the current pixel is not the old color
6) Change old color to new color
7) Check all the 4 directions

def flood_fill(image, sr, sc, new_color):
	old_color = image[sr][sc]
	if old_color == new_color:
		return image
	def dfs(row, col):
		if row < 0 or row >= len(image) or col < 0 or col >= len(image[0]):
			return
		
		if image[row][col] != old_color:
			return
		
		image[row][col] = new_color
		
		dfs(row - 1,col)
		dfs(row + 1,col)
		dfs(row,col - 1)
		dfs(row,col + 1)
		
	dfs(sr,sc)
	
	return image
	
image = [[1,1,1],[1,1,0], [1,0,1]]
sr = 1
sc = 1
new_color = 2
result = flood_fill(image, sr, sc, new_color)

8) Longest Increasing Subsequence (LIS) length.

def length_of_lis(nums):
    dp = [1] * len(nums)

    for i in range(len(nums)):
        for j in range(i):
            if nums[j] < nums[i]:
                dp[i] = max(dp[i], dp[j] + 1)

    return max(dp)


nums = [10, 9, 2, 5, 3, 7, 101, 18]

print(length_of_lis(nums))

def length_of_lis(nums):

    # Initially, every number itself has LIS length 1
    dp = [1] * len(nums)

    # i represents the current number
    for i in range(len(nums)):

        # j checks all the numbers before i
        for j in range(i):

            # Check if previous number is smaller than current number
            if nums[j] < nums[i]:

                # Extend the previous increasing subsequence by adding current number
                dp[i] = max(dp[i], dp[j] + 1)

    # Return the largest LIS length
    return max(dp)


# Input list
nums = [10, 9, 2, 5, 3, 7, 101, 18]

# Find and print the LIS length
print(length_of_lis(nums))

9) Coin Change: Given coins and amount, return minimum coins needed, else -1.

First, the question means:

coins = [1, 2, 5]
amount = 11

We need to make 11 using the minimum number of coins.

5 + 5 + 1 = 11

So the answer is 3 coins.

def coin_change(coins, amount):

    # dp[i] = minimum coins needed to make amount i
    dp = [amount + 1] * (amount + 1)

    # 0 coins are needed to make amount 0
    dp[0] = 0

    # Check every amount from 1 to target amount
    for i in range(1, amount + 1):

        # Try every coin
        for coin in coins:

            # Use coin only if it is not bigger than current amount
            if coin <= i:

                # Minimum of:
                # existing answer OR 1 coin + answer for remaining amount
                dp[i] = min(dp[i], 1 + dp[i - coin])

    # If amount cannot be made, return -1
    if dp[amount] == amount + 1:
        return -1

    return dp[amount]


coins = [1, 2, 5]
amount = 11

print(coin_change(coins, amount))

10) Minimum in Rotated Sorted Array: Return the minimum element in rotated sorted nums (no duplicates).

def find_min(nums):

    # Assume first number is minimum
    minimum = nums[0]

    # Check every number
    for num in nums:

        # If smaller number is found
        if num < minimum:
            minimum = num

    # Return minimum number
    return minimum


nums = [3, 4, 5, 1, 2]

print(find_min(nums))

11) Find Peak Element: Return any peak index where nums[i] > neighbors. 

def find_peak(nums):

    for i in range(len(nums) - 1):

        # Current number is greater than next number
        if nums[i] > nums[i + 1]:
            return i

    # If array keeps increasing, last element is peak
    return len(nums) - 1


nums = [1, 2, 3, 1]

def search(nums, target):

    # Check every element
    for i in range(len(nums)):

13) Rotate Array: Rotate nums to the right by k steps (in-place).

def rotate(nums, k):

    # Repeat k times
    for i in range(k):

        # Remove last element
        last = nums.pop()

        # Add it at the beginning
        nums.insert(0, last)

14) Kth Largest Element in an Array: Return kth largest element.

What does Kth Largest mean?

Suppose:

nums = [3, 2, 1, 5, 6, 4]
k = 2

Arrange the numbers from largest to smallest:

6, 5, 4, 3, 2, 1
↑  ↑
1st 2nd

So:

1st largest = 6
2nd largest = 5
3rd largest = 4

Since k = 2, the answer is:

5
-----------------------------------

def find_kth_largest(nums, k):

    # Sort numbers from smallest to largest
    nums.sort()

    # Return kth element from the end
    return nums[-k]


nums = [3, 2, 1, 5, 6, 4]
k = 2

print(find_kth_largest(nums, k))

15) Product of Array Except Self: Return array where ans[i] = product(nums) except nums[i], no division.

def product_except_self(nums):

    result = []

    # Pick each index
    for i in range(len(nums)):

        product = 1

        # Check every number
        for j in range(len(nums)):

            # Don't multiply the current number
            if i != j:
                product *= nums[j]

        result.append(product)

    return result


nums = [1, 2, 3, 4]

print(product_except_self(nums))

Remove duplicates from a list (preserve order)

def remove_duplicates(lst):
    seen = set()
    result = []
    
    for item in lst:
        if item not in seen:
            seen.add(item)
            result.append(item)
    return result
    
nums = [1,2,3,2,3]
print(remove_duplicates(nums))

    unique_nums = list(set(nums))
    unique_nums.sort()
    return unique_nums[-2]
